1//! Crate for creating sandbox environments to perform actions in isolation.
2//!
3//! This crate provides two main types:
4//! - [`Jailer`]: A simple sandbox that changes the current working directory
5//!   into a temporary one. When dropped or closed, it restores the original
6//!   directory and cleans up the temporary space.
7//! - [`EnvJailer`]: Extends [`Jailer`] by also managing environment variables,
8//!   allowing preservation of selected variables while clearing others on exit.
9//!
10//! Both types are thread-safe and ensure only one instance runs at a time via
11//! a global mutex.
12
13use std::collections::{HashMap, HashSet};
14use std::ffi::{OsStr, OsString};
15use std::path::PathBuf;
16use std::sync::{Arc, LazyLock};
17
18use parking_lot::lock_api::ArcMutexGuard;
19use parking_lot::{Mutex, RawMutex};
20use tempfile::TempDir;
21#[cfg(test)]
22use tokio_test as _;

24/// Global mutex which serializes jailer instances across threads
25static MUTEX: LazyLock<Arc<Mutex<()>>> = LazyLock::new(|| Arc::new(Mutex::new(())));
26
27/// [`Jailer`] struct which creates a jail environment.
28///
29/// [`Jailer`] creates a temporary directory and changes the current working
30/// directory to it. On drop or manual close, it restores the original working
31/// directory and deletes the temporary directory.
32///
33/// It uses a global mutex to ensure only one `Jailer` is active at a time
34/// across threads.
35pub struct Jailer {
36    temp_directory: Option<TempDir>,
37    original_directory: PathBuf,
38    _lock: ArcMutexGuard<RawMutex, ()>,
39    is_closed: bool,
40}
41
42impl Jailer {
43    /// Create a new [`Jailer`].
44    ///
45    /// This will:
46    /// - Lock globally to prevent concurrent instances.
47    /// - Create a temporary directory.
48    /// - Change the current directory to that temp dir.
49    ///
50    /// # Errors
51    ///
52    /// Returns an error if:
53    /// - The temporary directory cannot be created.
54    /// - Changing the current directory fails.
55    ///
56    /// # Deadlock
57    ///
58    /// Creating a second jailer while another one is still alive on the same
59    /// thread will deadlock, since the global mutex is not reentrant. Close
60    /// or drop the first jailer before creating a new one.
61    ///
62    /// # Example
63    ///
64    /// ```rust
65    /// use jailer::Jailer;
66    ///
67    /// // Capture the original working directory
68    /// let original_directory = std::env::current_dir().unwrap();
69    ///
70    /// let mut jailer = Jailer::new().unwrap();
71    ///
72    /// // We're now inside the jail
73    /// let inside_jailer_directory = std::env::current_dir().unwrap();
74    /// assert_ne!(inside_jailer_directory, original_directory);
75    ///
76    /// // Do some action in jailer...
77    ///
78    /// // Close the jail explicitly
79    /// jailer.close().unwrap();
80    ///
81    /// // Back to original directory
82    /// let after_jailer_directory = std::env::current_dir().unwrap();
83    /// assert_eq!(after_jailer_directory, original_directory);
84    /// assert_ne!(inside_jailer_directory, after_jailer_directory);
85    /// ```
86    pub fn new() -> Result<Self, std::io::Error> {
87        let lock = MUTEX.lock_arc();
88        let temp_dir = TempDir::new()?;
89        let original_directory = std::env::current_dir()?;
90        std::env::set_current_dir(&temp_dir)?;
91        Ok(Self {
92            temp_directory: Some(temp_dir),
93            original_directory,
94            _lock: lock,
95            is_closed: false,
96        })
97    }
98
99    /// Get a reference to the original directory
100    ///
101    /// This returns the directory that was active when the [`Jailer`] was
102    /// created
103    ///
104    /// # Example
105    ///
106    /// ```rust
107    /// use jailer::Jailer;
108    ///
109    /// let original_directory = std::env::current_dir().unwrap();
110    /// let jailer = Jailer::new().unwrap();
111    ///
112    /// assert_eq!(jailer.original_directory(), &original_directory);
113    /// jailer.close().unwrap();
114    /// ```
115    #[must_use]
116    pub fn original_directory(&self) -> &PathBuf {
117        &self.original_directory
118    }
119
120    /// Closes the [`Jailer`] and performs cleanup.
121    ///
122    /// This method:
123    /// - Changes back to the original working directory.
124    /// - Deletes the temporary directory.
125    /// - Releases the global lock.
126    ///
127    /// It consumes `self`, so the jailer cannot be used afterward.
128    ///
129    /// # Errors
130    ///
131    /// Returns an error if changing the directory or deleting the temp dir
132    /// fails.
133    pub fn close(mut self) -> Result<(), std::io::Error> {
134        std::env::set_current_dir(self.original_directory.as_path())?;
135        if let Some(temp) = self.temp_directory.take() {
136            temp.close()?;
137        }
138        self.is_closed = true;
139        Ok(())
140    }
141}
142
143impl Drop for Jailer {
144    fn drop(&mut self) {
145        if !self.is_closed {
146            std::env::set_current_dir(self.original_directory.as_path()).ok();
147            if let Some(temp) = self.temp_directory.take() {
148                temp.close().ok();
149            }
150        }
151    }
152}

154/// [`EnvJailer`] struct which creates a jail environment with environment
155/// variable management.
156///
157/// [`EnvJailer`] wraps [`Jailer`] and adds support for preserving specific
158/// environment variables. On drop or close, it:
159/// - Removes all environment variables not marked as preserved.
160/// - Restores original values for preserved keys.
161/// - Reverts to the original working directory and cleans up the temp dir.
162///
163/// # Safety
164///
165/// Environment variable manipulation via [`std::env::set_var`] and
166/// [`std::env::remove_var`] is considered unsafe due to potential race
167/// conditions in multi-threaded programs. Therefore, methods like
168/// [`EnvJailer::close`] are marked as `unsafe`.
169pub struct EnvJailer {
170    jailer: Option<Jailer>,
171    original_directory: PathBuf,
172    original_env_vars_os: HashMap<OsString, OsString>,
173    preserved_env_vars_os: HashSet<OsString>,
174}
175
176impl EnvJailer {
177    /// Create a new [`EnvJailer`].
178    ///
179    /// This captures the current environment variables and working directory,
180    /// then initializes a new [`Jailer`].
181    ///
182    /// # Errors
183    ///
184    /// Returns an error if the underlying [`Jailer`] cannot be created.
185    ///
186    /// # Example
187    ///
188    /// ```rust
189    /// use jailer::EnvJailer;
190    ///
191    /// // Capture the original working directory
192    /// let original_directory = std::env::current_dir().unwrap();
193    ///
194    /// let mut env_jailer = EnvJailer::new().unwrap();
195    ///
196    /// // Do some action in jailer...
197    ///
198    /// // Close the jail explicitly (unsafe due to env var changes)
199    /// unsafe {
200    ///     env_jailer.close().unwrap();
201    /// }
202    ///
203    /// // Back to original directory
204    /// assert_eq!(std::env::current_dir().unwrap(), original_directory);
205    /// ```
206    pub fn new() -> Result<Self, std::io::Error> {
207        // create jailer first so global lock is held while capturing env vars
208        // and original dir
209        let jailer = Jailer::new()?;
210        let original_env_vars_os = std::env::vars_os().collect();
211        let original_dir = jailer.original_directory().clone();
212
213        Ok(Self {
214            jailer: Some(jailer),
215            original_directory: original_dir,
216            original_env_vars_os,
217            preserved_env_vars_os: HashSet::new(),
218        })
219    }
220
221    /// Get a reference to the original directory
222    ///
223    /// This returns the directory that was active when the [`EnvJailer`] was
224    /// created
225    ///
226    /// # Example
227    ///
228    /// ```rust
229    /// use jailer::EnvJailer;
230    ///
231    /// let original_directory = std::env::current_dir().unwrap();
232    /// let env_jailer = EnvJailer::new().unwrap();
233    /// assert_eq!(env_jailer.original_directory(), &original_directory);
234    /// unsafe {
235    ///     env_jailer.close().unwrap();
236    /// }
237    /// ```
238    #[must_use]
239    pub fn original_directory(&self) -> &PathBuf {
240        &self.original_directory
241    }
242
243    /// Mark an environment variable as preserved.
244    ///
245    /// When the jailer is closed or dropped, this key will retain its current
246    /// value instead of being removed or reset to the original value.
247    ///
248    /// # Example
249    ///
250    /// ```rust
251    /// use jailer::EnvJailer;
252    ///
253    /// unsafe {
254    ///     std::env::set_var("KEY", "VALUE");
255    ///     std::env::set_var("ANOTHER_KEY", "VALUE");
256    /// }
257    ///
258    /// let mut env_jailer = EnvJailer::new().unwrap();
259    ///
260    /// assert_eq!(std::env::var("KEY"), Ok("VALUE".to_string()));
261    ///
262    /// unsafe {
263    ///     std::env::set_var("KEY2", "VALUE2");
264    /// }
265    /// env_jailer.set_preserved_env("KEY");
266    ///
267    /// unsafe {
268    ///     std::env::set_var("KEY", "VALUE2");
269    ///     std::env::set_var("ANOTHER_KEY", "ANOTHER_VAL");
270    /// }
271    ///
272    /// assert_eq!(std::env::var("KEY"), Ok("VALUE2".to_string()));
273    /// assert_eq!(std::env::var("ANOTHER_KEY"), Ok("ANOTHER_VAL".to_string()));
274    ///
275    /// unsafe {
276    ///     env_jailer.close().unwrap();
277    /// }
278    ///
279    /// assert_eq!(std::env::var("KEY"), Ok("VALUE2".to_string()));
280    /// assert_eq!(std::env::var("ANOTHER_KEY"), Ok("VALUE".to_string()));
281    /// assert!(std::env::var("KEY2").is_err());
282    /// ```
283    pub fn set_preserved_env<K>(&mut self, key: K)
284    where
285        K: AsRef<OsStr>,
286    {
287        self.preserved_env_vars_os
288            .insert(key.as_ref().to_os_string());
289    }
290
291    /// Remove an environment variable from the preserved list.
292    ///
293    /// Note: This does *not* remove the current environment variable.
294    /// To remove it, call [`std::env::remove_var`] manually.
295    ///
296    /// # Example
297    ///
298    /// ```rust
299    /// use jailer::EnvJailer;
300    ///
301    /// let mut env_jailer = EnvJailer::new().unwrap();
302    ///
303    /// unsafe {
304    ///     std::env::set_var("KEY", "VALUE");
305    /// }
306    ///
307    /// env_jailer.set_preserved_env("KEY");
308    /// assert_eq!(std::env::var("KEY"), Ok("VALUE".to_string()));
309    /// env_jailer.remove_preserved_env("KEY");
310    ///
311    /// unsafe {
312    ///     env_jailer.close().unwrap();
313    /// }
314    ///
315    /// assert!(std::env::var("KEY").is_err());
316    /// ```
317    pub fn remove_preserved_env<K>(&mut self, key: K)
318    where
319        K: AsRef<OsStr>,
320    {
321        self.preserved_env_vars_os.remove(key.as_ref());
322    }
323
324    /// Returns a reference to the map of original environment variables.
325    ///
326    /// These are the environment variables present when the [`EnvJailer`] was
327    /// created. Any variables added during the session are not included
328    /// here.
329    #[must_use]
330    pub fn original_env_vars_os(&self) -> &HashMap<OsString, OsString> {
331        &self.original_env_vars_os
332    }
333
334    /// Returns a reference to the set of preserved environment variable names.
335    #[must_use]
336    pub fn preserved_env_vars_os(&self) -> &HashSet<OsString> {
337        &self.preserved_env_vars_os
338    }
339
340    unsafe fn revert_env_vars(&self) {
341        let current_keys: Vec<OsString> = std::env::vars_os().map(|(key, _)| key).collect();
342        for key in current_keys {
343            if !self.preserved_env_vars_os.contains(&key) {
344                unsafe {
345                    std::env::remove_var(key);
346                }
347            }
348        }
349        for (key, value) in &self.original_env_vars_os {
350            if !self.preserved_env_vars_os.contains(key) {
351                unsafe {
352                    std::env::set_var(key, value);
353                }
354            }
355        }
356    }
357
358    /// Closes the [`EnvJailer`] and performs cleanup.
359    ///
360    /// This method:
361    /// - Reverts environment variables to their original state (except
362    ///   preserved ones).
363    /// - Closes the underlying [`Jailer`] (restoring directory and removing
364    ///   temp dir).
365    ///
366    /// It consumes `self`, so the jailer cannot be used afterward.
367    ///
368    /// # Errors
369    ///
370    /// Returns an error if the underlying [`Jailer::close`] fails.
371    ///
372    /// # Safety
373    ///
374    /// This function calls [`std::env::remove_var`] and [`std::env::set_var`],
375    /// which are unsafe due to possible data races in concurrent contexts.
376    pub unsafe fn close(mut self) -> Result<(), std::io::Error> {
377        unsafe {
378            self.revert_env_vars();
379        }
380        if let Some(jailer) = self.jailer.take() {
381            jailer.close()?;
382        }
383        Ok(())
384    }
385}
386
387impl Drop for EnvJailer {
388    fn drop(&mut self) {
389        if self.jailer.is_some() {
390            unsafe {
391                self.revert_env_vars();
392            }
393        }
394    }
395}

397/// Run a closure inside a [`Jailer`] environment.
398///
399/// This function creates a [`Jailer`], runs the provided closure, and ensures
400/// the jail is closed afterward.
401///
402/// # Errors
403///
404/// Returns any error from creating or closing the [`Jailer`], or from the
405/// closure itself.
406///
407/// # Example
408///
409/// ```rust
410/// use jailer::with_jailer;
411///
412/// with_jailer(|_jailer| {
413///     // Do something in the jailed directory
414///     Ok(())
415/// })
416/// .unwrap();
417/// ```
418pub fn with_jailer<F, T>(f: F) -> Result<T, Box<dyn std::error::Error>>
419where
420    F: FnOnce(&Jailer) -> Result<T, Box<dyn std::error::Error>>,
421{
422    let jailer = Jailer::new()?;
423    let result = f(&jailer);
424    jailer.close()?;
425    result
426}

428/// Run a closure inside a [`Jailer`] environment asynchronously.
429///
430/// This function creates a [`Jailer`], runs the provided async closure, and
431/// ensures the jail is closed afterward.
432///
433/// # Errors
434///
435/// Returns any error from creating or closing the [`Jailer`], or from the
436/// closure itself.
437///
438/// # Example
439///
440/// ```rust
441/// use jailer::with_jailer_async;
442/// # tokio_test::block_on(async {
443/// with_jailer_async(|_jailer| {
444///     async {
445///         // Do something in the jailed directory
446///         Ok(())
447///     }
448/// })
449/// .await
450/// .unwrap();
451/// # })
452/// ```
453pub async fn with_jailer_async<F, T, Fut>(f: F) -> Result<T, Box<dyn std::error::Error>>
454where
455    F: FnOnce(&Jailer) -> Fut,
456    Fut: Future<Output = Result<T, Box<dyn std::error::Error>>>,
457{
458    let jailer = Jailer::new()?;
459    let result = f(&jailer).await;
460    jailer.close()?;
461    result
462}

464/// Run a closure inside a [`EnvJailer`] environment.
465///
466/// This function creates a [`EnvJailer`], runs the provided closure, and
467/// ensures the jail is closed afterward.
468///
469/// # Errors
470///
471/// Returns any error from creating or closing the [`EnvJailer`], or from the

473///
474/// # Example
475///
476/// ```rust
477/// use jailer::with_env_jailer;
478/// unsafe {
479///     with_env_jailer(|env_jailer| {

481///         std::env::set_var("JAILED_ONLY_KEY", "VALUE");
482///         env_jailer.set_preserved_env("SOME_KEY");
483///         Ok(())
484///     })
485///     .unwrap();
486/// }

488/// // Variables set inside the jail are reverted on close
489/// assert!(std::env::var("JAILED_ONLY_KEY").is_err());

491///

493///
494/// This function calls [`std::env::remove_var`] and [`std::env::set_var`],
495/// which are unsafe due to possible data races in concurrent contexts.
496pub unsafe fn with_env_jailer<F, T>(f: F) -> Result<T, Box<dyn std::error::Error>>
497where
498    F: FnOnce(&mut EnvJailer) -> Result<T, Box<dyn std::error::Error>>,
499{
500    let mut env_jailer = EnvJailer::new()?;
501    let result = f(&mut env_jailer);
502    unsafe {
503        env_jailer.close()?;
504    }
505    result
506}
